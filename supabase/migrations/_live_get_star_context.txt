CREATE OR REPLACE FUNCTION public.get_star_context(p_tenant_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
DECLARE
  v_result JSONB;
BEGIN
  SELECT jsonb_build_object(

    'generated_at', NOW(),
    'tenant_id', p_tenant_id,

    -- STUDENTS
    'students', jsonb_build_object(
      'active',   (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND status = 'active'),
      'paused',   (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND status = 'paused'),
      'inactive', (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND status IN ('inactive','former')),
      'total',    (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id),
      'new_last_30_days', (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND created_at >= NOW() - INTERVAL '30 days'),
      'new_last_7_days',  (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND created_at >= NOW() - INTERVAL '7 days'),
      'military_count',   (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND status = 'active' AND is_military = true),
      'at_risk_no_lesson_30_days', (
        SELECT COUNT(*) FROM students s
        WHERE s.tenant_id = p_tenant_id AND s.status = 'active'
        AND NOT EXISTS (SELECT 1 FROM session_log sl WHERE sl.student_id = s.id AND sl.block_date >= CURRENT_DATE - 30)
      ),
      'by_location', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', location, 'count', student_count) ORDER BY student_count DESC), '[]')
        FROM (
          SELECT l.name AS location, COUNT(s.id) AS student_count
          FROM students s JOIN locations l ON s.location_id = l.id
          WHERE s.tenant_id = p_tenant_id AND s.status = 'active'
          GROUP BY l.name
        ) t
      ),
      'by_instrument', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('instrument', instrument, 'count', cnt) ORDER BY cnt DESC), '[]')
        FROM (
          SELECT INITCAP(LOWER(instrument)) AS instrument, COUNT(*) AS cnt
          FROM students
          WHERE tenant_id = p_tenant_id AND status = 'active' AND instrument IS NOT NULL AND instrument != ''
          GROUP BY INITCAP(LOWER(instrument))
          ORDER BY cnt DESC LIMIT 15
        ) t
      )
    ),

    -- FAMILIES
    'families', jsonb_build_object(
      'total',               (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id),
      'no_card_on_file',     (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND card_last_four IS NULL),
      'with_card_on_file',   (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND card_last_four IS NOT NULL),
      'autopay_enabled',     (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND autopay_enabled = true),
      'military',            (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND is_military = true),
      'total_overdue_cents', (SELECT COALESCE(SUM(overdue_balance_cents), 0) FROM families WHERE tenant_id = p_tenant_id),
      'families_overdue',    (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND overdue_balance_cents > 0),
      'sms_opted_out',       (SELECT COUNT(*) FROM families WHERE tenant_id = p_tenant_id AND sms_opted_out = true)
    ),

    -- BILLING / REVENUE
    'billing', jsonb_build_object(
      'estimated_mrr_cents', (
        SELECT COALESCE(SUM(monthly_cents), 0) FROM student_effective_rate
        WHERE tenant_id = p_tenant_id AND status = 'active'
      ),
      'rate_tiers', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('rate_tier_cents', rate_tier, 'family_count', cnt) ORDER BY rate_tier DESC), '[]')
        FROM (
          SELECT rate_tier, COUNT(*) AS cnt FROM families
          WHERE tenant_id = p_tenant_id AND rate_tier IS NOT NULL
          GROUP BY rate_tier
        ) t
      ),
      'mrr_by_location', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', loc_name, 'mrr_cents', loc_mrr) ORDER BY loc_mrr DESC), '[]')
        FROM (
          SELECT l.name AS loc_name, COALESCE(SUM(ser.monthly_cents), 0) AS loc_mrr
          FROM locations l
          LEFT JOIN student_effective_rate ser ON ser.location_id = l.id AND ser.tenant_id = p_tenant_id AND ser.status = 'active'
          WHERE l.tenant_id = p_tenant_id AND l.is_active = true
          GROUP BY l.name
        ) t
      ),
      'latest_billing_cycle', (
        SELECT jsonb_build_object(
          'label', label,
          'status', status,
          'total_base_cents', total_base_cents,
          'total_adjusted_cents', total_adjusted_cents,
          'total_paid_cents', total_paid_cents,
          'billing_month', billing_month
        )
        FROM billing_cycles
        WHERE tenant_id = p_tenant_id
        ORDER BY billing_month DESC LIMIT 1
      )
    ),

    -- TEACHERS
    'teachers', jsonb_build_object(
      'active',   (SELECT COUNT(*) FROM teachers WHERE tenant_id = p_tenant_id AND is_active = true),
      'inactive', (SELECT COUNT(*) FROM teachers WHERE tenant_id = p_tenant_id AND is_active = false),
      'no_students', (
        SELECT COUNT(*) FROM teachers t
        WHERE t.tenant_id = p_tenant_id AND t.is_active = true
        AND NOT EXISTS (SELECT 1 FROM students s WHERE s.teacher_id = t.id AND s.status = 'active')
      ),
      'contract_missing', (
        SELECT COUNT(*) FROM teachers
        WHERE tenant_id = p_tenant_id AND is_active = true
        AND (contract_status IS NULL OR contract_status NOT IN ('signed','complete'))
      ),
      'w9_missing', (
        SELECT COUNT(*) FROM teachers
        WHERE tenant_id = p_tenant_id AND is_active = true AND needs_1099 = true
        AND (w9_status IS NULL OR w9_status != 'complete')
      ),
      'load_by_teacher', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object(
          'name', teacher_name,
          'active_students', student_count,
          'instruments', instruments
        ) ORDER BY student_count DESC), '[]')
        FROM (
          SELECT
            t.first_name || ' ' || t.last_name AS teacher_name,
            COUNT(s.id) AS student_count,
            t.instruments
          FROM teachers t
          LEFT JOIN students s ON s.teacher_id = t.id AND s.status = 'active'
          WHERE t.tenant_id = p_tenant_id AND t.is_active = true
          GROUP BY t.id, t.first_name, t.last_name, t.instruments
        ) t
      ),
      'by_location', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', loc_name, 'teacher_count', cnt) ORDER BY cnt DESC), '[]')
        FROM (
          SELECT l.name AS loc_name, COUNT(DISTINCT pl.profile_id) AS cnt
          FROM profile_locations pl
          JOIN locations l ON l.id = pl.location_id
          WHERE l.tenant_id = p_tenant_id AND l.is_active = true
          GROUP BY l.name
        ) t
      )
    ),

    -- SCHEDULE
    'schedule', jsonb_build_object(
      'booked_this_week', (
        SELECT COUNT(*) FROM schedule_blocks
        WHERE tenant_id = p_tenant_id AND block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6 AND status = 'booked'
      ),
      'available_this_week', (
        SELECT COUNT(*) FROM schedule_blocks
        WHERE tenant_id = p_tenant_id AND block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6 AND status = 'available'
      ),
      'booked_this_month', (
        SELECT COUNT(*) FROM schedule_blocks
        WHERE tenant_id = p_tenant_id
        AND block_date BETWEEN DATE_TRUNC('month', CURRENT_DATE) AND (DATE_TRUNC('month', CURRENT_DATE) + INTERVAL '1 month' - INTERVAL '1 day')
        AND status = 'booked'
      ),
      'callouts_this_week', (
        SELECT COUNT(*) FROM schedule_blocks
        WHERE tenant_id = p_tenant_id AND block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6 AND block_type = 'call_out'
      ),
      'utilization_pct', (
        SELECT CASE WHEN (booked + avail) = 0 THEN 0
          ELSE ROUND(booked::numeric / (booked + avail) * 100, 1) END
        FROM (
          SELECT
            COUNT(CASE WHEN status = 'booked' THEN 1 END) AS booked,
            COUNT(CASE WHEN status = 'available' THEN 1 END) AS avail
          FROM schedule_blocks
          WHERE tenant_id = p_tenant_id AND block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6
        ) t
      ),
      'by_location_this_week', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', loc_name, 'booked', booked, 'available', avail) ORDER BY booked DESC), '[]')
        FROM (
          SELECT l.name AS loc_name,
            COUNT(CASE WHEN sb.status = 'booked' THEN 1 END) AS booked,
            COUNT(CASE WHEN sb.status = 'available' THEN 1 END) AS avail
          FROM schedule_blocks sb JOIN locations l ON l.id = sb.location_id
          WHERE sb.tenant_id = p_tenant_id AND sb.block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6
          GROUP BY l.name
        ) t
      )
    ),

    -- LEADS
    'leads', jsonb_build_object(
      'active_total',          (SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND stage NOT IN ('enrolled','lost')),
      'needing_followup',      (SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND stage NOT IN ('enrolled','lost') AND (next_follow_up_at < NOW() OR next_follow_up_at IS NULL)),
      'new_last_7_days',       (SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND created_at >= NOW() - INTERVAL '7 days'),
      'new_last_30_days',      (SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND created_at >= NOW() - INTERVAL '30 days'),
      'converted_last_30_days',(SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND stage = 'enrolled' AND updated_at >= NOW() - INTERVAL '30 days'),
      'lost_last_30_days',     (SELECT COUNT(*) FROM leads WHERE tenant_id = p_tenant_id AND stage = 'lost' AND updated_at >= NOW() - INTERVAL '30 days'),
      'by_stage', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('stage', stage::text, 'count', cnt)), '[]')
        FROM (SELECT stage, COUNT(*) AS cnt FROM leads WHERE tenant_id = p_tenant_id AND stage != 'lost' GROUP BY stage) t
      ),
      'by_instrument', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('instrument', instrument, 'count', cnt) ORDER BY cnt DESC), '[]')
        FROM (
          SELECT instrument, COUNT(*) AS cnt FROM leads
          WHERE tenant_id = p_tenant_id AND stage NOT IN ('enrolled','lost') AND instrument IS NOT NULL
          GROUP BY instrument ORDER BY cnt DESC
        ) t
      ),
      'by_location', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', loc_name, 'count', cnt) ORDER BY cnt DESC), '[]')
        FROM (
          SELECT l.name AS loc_name, COUNT(*) AS cnt
          FROM leads ld JOIN locations l ON l.id = ld.location_id
          WHERE ld.tenant_id = p_tenant_id AND ld.stage NOT IN ('enrolled','lost')
          GROUP BY l.name ORDER BY cnt DESC
        ) t
      )
    ),

    -- SESSIONS
    'sessions', jsonb_build_object(
      'total_last_30_days', (SELECT COUNT(*) FROM session_log WHERE tenant_id = p_tenant_id AND block_date >= CURRENT_DATE - 30),
      'total_last_7_days',  (SELECT COUNT(*) FROM session_log WHERE tenant_id = p_tenant_id AND block_date >= CURRENT_DATE - 7),
      'notes_written_last_7_days', (
        SELECT COUNT(*) FROM session_log
        WHERE tenant_id = p_tenant_id AND block_date >= CURRENT_DATE - 7 AND lesson_notes IS NOT NULL AND lesson_notes != ''
      ),
      'avg_engagement_last_30_days', (
        SELECT ROUND(AVG(engagement_level)::numeric, 1)
        FROM session_log WHERE tenant_id = p_tenant_id AND block_date >= CURRENT_DATE - 30 AND engagement_level IS NOT NULL
      ),
      'by_location_last_30_days', (
        SELECT COALESCE(jsonb_agg(jsonb_build_object('location', loc_name, 'sessions', cnt) ORDER BY cnt DESC), '[]')
        FROM (
          SELECT l.name AS loc_name, COUNT(*) AS cnt
          FROM session_log sl JOIN locations l ON l.id = sl.location_id
          WHERE sl.tenant_id = p_tenant_id AND sl.block_date >= CURRENT_DATE - 30
          GROUP BY l.name ORDER BY cnt DESC
        ) t
      )
    ),

    -- RETENTION
    'retention', jsonb_build_object(
      'students_paused',             (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND status = 'paused'),
      'students_may_return',         (SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id AND may_return = 'yes'),
      'students_inactive_last_60_days', (
        SELECT COUNT(*) FROM students WHERE tenant_id = p_tenant_id
        AND status IN ('inactive','former') AND deactivated_at >= NOW() - INTERVAL '60 days'
      ),
      'active_campaigns',            (SELECT COUNT(*) FROM retention_campaigns WHERE tenant_id = p_tenant_id AND status = 'pending'),
      'sent_campaigns_last_30_days', (SELECT COUNT(*) FROM retention_campaigns WHERE tenant_id = p_tenant_id AND sent_at >= NOW() - INTERVAL '30 days')
    ),

    -- TASKS
    'tasks', jsonb_build_object(
      'open',               (SELECT COUNT(*) FROM tasks WHERE tenant_id = p_tenant_id AND status = 'open'),
      'overdue',            (SELECT COUNT(*) FROM tasks WHERE tenant_id = p_tenant_id AND status = 'open' AND due_date < CURRENT_DATE),
      'high_priority_open', (SELECT COUNT(*) FROM tasks WHERE tenant_id = p_tenant_id AND status = 'open' AND priority = 'high')
    ),

    -- LOCATIONS (per-location breakdown)
    'locations', (
      SELECT COALESCE(jsonb_agg(jsonb_build_object(
        'id', l.id,
        'name', l.name,
        'active_students',   (SELECT COUNT(*) FROM students s WHERE s.location_id = l.id AND s.status = 'active'),
        'active_teachers',   (SELECT COUNT(DISTINCT t.id) FROM teachers t JOIN profile_locations pl ON pl.profile_id = t.profile_id WHERE pl.location_id = l.id AND t.is_active = true),
        'booked_this_week',  (SELECT COUNT(*) FROM schedule_blocks sb WHERE sb.location_id = l.id AND sb.block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6 AND sb.status = 'booked'),
        'available_this_week',(SELECT COUNT(*) FROM schedule_blocks sb WHERE sb.location_id = l.id AND sb.block_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 6 AND sb.status = 'available'),
        'mrr_cents',         (SELECT COALESCE(SUM(ser.monthly_cents),0) FROM student_effective_rate ser WHERE ser.location_id = l.id AND ser.tenant_id = p_tenant_id AND ser.status = 'active')
      ) ORDER BY l.name), '[]')
      FROM locations l WHERE l.tenant_id = p_tenant_id AND l.is_active = true
    ),

    -- COMMUNICATIONS
    'communications', jsonb_build_object(
      'sent_last_7_days',  (SELECT COUNT(*) FROM communications WHERE tenant_id = p_tenant_id AND sent_at >= NOW() - INTERVAL '7 days'),
      'sent_last_30_days', (SELECT COUNT(*) FROM communications WHERE tenant_id = p_tenant_id AND sent_at >= NOW() - INTERVAL '30 days')
    ),

    -- PAYROLL
    'payroll', jsonb_build_object(
      'latest_period', (
        SELECT jsonb_build_object(
          'period_id',   pp.id,
          'period_label', pp.period_label,
          'start_date',  pp.start_date,
          'end_date',    pp.end_date,
          'is_closed',   pp.is_closed
        )
        FROM payroll_periods pp
        WHERE pp.tenant_id = p_tenant_id
        ORDER BY pp.created_at DESC LIMIT 1
      )
    )

  ) INTO v_result;

  RETURN v_result;
END;
$function$
